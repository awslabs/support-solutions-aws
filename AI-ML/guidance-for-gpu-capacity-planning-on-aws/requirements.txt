# Pinned to specific versions to close a supply-chain gap flagged in a
# security review. The previous `>=1.34.0` floor let any
# fresh deploy pull the current PyPI HEAD at build time.
#
# When upgrading:
#   1. Bump both boto3 and botocore in lockstep (they're released together).
#   2. Re-run `make ci` — targets zero lint / test / SAM validate errors.
#   3. Commit the version bump as its own `chore(deps): ...` change so
#      the audit trail stays clean.
#
# Version rationale: 1.42.97 is the last version tested end-to-end
# (as of 2026-07-10). AWS releases boto3 / botocore roughly
# every business day; adopt new versions deliberately, not by drift.
boto3==1.42.97
botocore==1.42.97

# urllib3 is a transitive dependency of botocore. pip-audit flagged CVEs in the
# 1.26.x line that botocore pulls on Python < 3.10. The Lambda runtime is
# Python 3.13, where botocore permits urllib3 <3; pin to a patched line there to
# close those CVEs. The python_version marker avoids a conflict with botocore's
# <1.27 cap on older local interpreters (the deployed artifact is what matters).
urllib3>=2.8.0,<3; python_version >= "3.10"
